// -----------------------------
// 3.2 Función no recursiva con parámetros
// -----------------------------
func calcularVolumenPiramide(base float32, altura float32) float32 {
    return (base * altura) / 3.0
}

// -----------------------------
// 3.3 Funciones por referencia
// -----------------------------
func intercambioValores(a *int32, b *int32) {
    temp := *a
    *a = *b
    *b = temp
}

func ordenamientoSeleccion(arr *[6]int32) {
    for i := 0; i < 5; i++ {
        var minIdx int32 = i
        for j := i + 1; j < 6; j++ {
            if arr[j] < arr[minIdx] {
                minIdx = j
            }
        }
        if minIdx != i {
            intercambioValores(&arr[i], &arr[minIdx])
        }
    }
}

// -----------------------------
// 3.4 Múltiple retorno no recursiva
// -----------------------------
func division(a int32, b int32) (int32, bool) {
    if b == 0 {
        return 0, false
    }
    return a / b, true
}

// -----------------------------
// 3.4 Recursiva un retorno
// -----------------------------
func potencia(base int32, exp int32) int32 {
    if exp == 0 {
        return 1
    }
    return base * potencia(base, exp-1)
}

// -----------------------------
// 3.5 Recursiva múltiple retorno (Euclides)
// -----------------------------
func euclides(a int32, b int32) (int32, int32) {
    if b == 0 {
        return a, 0
    }

    gcd, depth := euclides(b, a%b)
    return gcd, depth + 1
}

// -----------------------------
// 5.6 Índice de inestabilidad por fila (matriz)
// Suma de diferencias absolutas consecutivas por fila
// -----------------------------
func indiceInestabilidadPorFila(m [3][4]int32) [3]int32 {
    var res [3]int32
    for i := 0; i < 3; i++ {
        var acum int32 = 0
        for j := 0; j < 3; j++ {
            var diff int32 = m[i][j+1] - m[i][j]
            if diff < 0 {
                diff = -diff
            }
            acum += diff
        }
        res[i] = acum
    }
    return res
}

// -----------------------------
// 5.7 Regla de Cramer (2x2)
// Retorna (x, y, ok)
// -----------------------------
func reglaCramer2x2(a11 float32, a12 float32, b1 float32, a21 float32, a22 float32, b2 float32) (float32, float32, bool) {
    var det float32 = a11*a22 - a12*a21
    if det == 0.0 {
        return 0.0, 0.0, false
    }

    var dx float32 = b1*a22 - a12*b2
    var dy float32 = a11*b2 - b1*a21
    return dx/det, dy/det, true
}

// -----------------------------
// 5.8 Promedio de capas (cubo 2x2x2)
// -----------------------------
func promedioCapas(cubo [2][2][2]float32) [2]float32 {
    var proms [2]float32
    for z := 0; z < 2; z++ {
        var sum float32 = 0.0
        for i := 0; i < 2; i++ {
            for j := 0; j < 2; j++ {
                sum += cubo[z][i][j]
            }
        }
        proms[z] = sum / 4.0
    }
    return proms
}

// -----------------------------
// 5.9 Softmax (aprox) por fila de matriz 2x2
// Usamos x^2 como aproximación positiva para evitar exp()
// -----------------------------
func softmaxAprox2x2(m [2][2]float32) [2][2]float32 {
    var out [2][2]float32
    for i := 0; i < 2; i++ {
        var a float32 = m[i][0] * m[i][0]
        var b float32 = m[i][1] * m[i][1]
        var s float32 = a + b

        if s == 0.0 {
            out[i][0] = 0.5
            out[i][1] = 0.5
        } else {
            out[i][0] = a / s
            out[i][1] = b / s
        }
    }
    return out
}
