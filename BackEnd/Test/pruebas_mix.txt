func sideEffect(tag string) bool {
    print("SIDE_EFFECT:", tag)
    return true
}

func swap(a *int32, b *int32) {
    t := *a
    *a = *b
    *b = t
}

func burbuja(arr *[20]int32) {
    for i := 0; i < 19; i++ {
        for j := 0; j < 19-i; j++ {
            if arr[j] > arr[j+1] {
                swap(&arr[j], &arr[j+1])
            }
        }
    }
}

func fibonacciN(n int32) int32 {
    if n <= 1 {
        return n
    }
    var a int32 = 0
    var b int32 = 1
    for i := 2; i <= n; i++ {
        var t int32 = a + b
        a = b
        b = t
    }
    return b
}

func factorial(n int32) int32 {
    if n <= 1 {
        return 1
    }
    return n * factorial(n-1)
}

func potencia(base int32, exp int32) int32 {
    if exp == 0 {
        return 1
    }
    return base * potencia(base, exp-1)
}

func mcd(a int32, b int32) int32 {
    if b == 0 {
        return a
    }
    return mcd(b, a%b)
}

func ackermann(m int32, n int32) int32 {
    if m == 0 {
        return n + 1
    }
    if n == 0 {
        return ackermann(m-1, 1)
    }
    return ackermann(m-1, ackermann(m, n-1))
}

func sumaMatriz5x5(m [5][5]int32) int32 {
    var s int32 = 0
    for i := 0; i < 5; i++ {
        for j := 0; j < 5; j++ {
            s += m[i][j]
        }
    }
    return s
}

func sumaCubo2(c [2][2][2]int32) int32 {
    var s int32 = 0
    for z := 0; z < 2; z++ {
        for i := 0; i < 2; i++ {
            for j := 0; j < 2; j++ {
                s += c[z][i][j]
            }
        }
    }
    return s
}
