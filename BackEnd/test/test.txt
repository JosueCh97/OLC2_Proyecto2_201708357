fmt.Print("--- 1. FOR CLASICO (con continue) ---")
for var i int32 = 0; i < 4; i++ {
    if i == 2 {
        fmt.Print("Saltando el: ", i)
        continue // Debe omitir imprimir el 2
    }
    fmt.Print("Iteracion: ", i)
}

fmt.Print("--- 2. FOR ESTILO WHILE ---")
var x int32 = 3
for x > 0 {
    fmt.Print("X vale: ", x)
    x--
}

fmt.Print("--- 3. FOR INFINITO (con break) ---")
var count int32 = 1
for {
    fmt.Print("Infinito count: ", count)
    if count == 2 {
        fmt.Print("Rompiendo ciclo infinito!")
        break // Si esto falla, tu programa se colgará (¡cuidado!)
    }
    count++
}
